cmake_minimum_required(VERSION 3.24)
# Select the deployment GPU explicitly when building on a machine without CUDA.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "Target CUDA architectures")
endif()
project(ppocrv6_native VERSION 0.1.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
# Only core/imgproc are needed; the full OpenCV development package is optional.
find_path(OPENCV_INCLUDE_DIR opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
find_library(OPENCV_CORE opencv_core REQUIRED)
find_library(OPENCV_IMGPROC opencv_imgproc REQUIRED)
find_path(CLIPPER_INCLUDE_DIR NAMES polyclipping/clipper.hpp clipper.hpp REQUIRED)
find_library(CLIPPER_LIBRARY polyclipping REQUIRED)
add_library(ppocrv6_core STATIC
  src/detection/db_postprocess.cpp src/recognition/ctc_decode.cpp
  src/recognition/glyph_shape.cpp)
target_include_directories(ppocrv6_core PUBLIC include ${OPENCV_INCLUDE_DIR} ${CLIPPER_INCLUDE_DIR})
target_link_libraries(ppocrv6_core PUBLIC ${OPENCV_CORE} ${OPENCV_IMGPROC} ${CLIPPER_LIBRARY})
add_library(ppocrv6_cuda STATIC src/kernels/classifier.cu
  src/kernels/detector_preprocess.cu src/kernels/roi_warp.cu)
target_include_directories(ppocrv6_cuda PUBLIC include)
target_link_libraries(ppocrv6_cuda PUBLIC ppocrv6_core CUDA::cudart)
target_compile_options(ppocrv6_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
add_library(ppocrv6_runtime STATIC src/engine/model_engine_compiled.cpp
  src/recognition/recognition_worker.cpp src/detection/detection_worker.cpp
  src/full_page/full_page_worker.cpp)
target_link_libraries(ppocrv6_runtime PUBLIC ppocrv6_cuda ${CMAKE_DL_LIBS} Threads::Threads)
add_library(ppocrv6_native SHARED src/c_api/common.cpp src/c_api/detector_c_api.cpp
  src/c_api/recognizer_c_api.cpp src/c_api/full_page_c_api.cpp)
target_link_libraries(ppocrv6_native PRIVATE ppocrv6_runtime)
install(TARGETS ppocrv6_native LIBRARY DESTINATION lib)
option(BUILD_TESTING "Build development tests" OFF)
if(BUILD_TESTING)
  enable_testing()
  foreach(test ctc_smoke glyph_shape_smoke)
    add_executable(${test} tests/tools/${test}.cpp)
    target_link_libraries(${test} PRIVATE ppocrv6_core)
    add_test(NAME ${test} COMMAND ${test})
  endforeach()
  foreach(test model_engine_smoke compiled_memory_smoke)
    add_executable(${test} tests/tools/${test}.cpp)
    target_link_libraries(${test} PRIVATE ppocrv6_runtime)
  endforeach()
endif()
